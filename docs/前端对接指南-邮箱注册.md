```typescript
// 发送验证码请求
export interface SendVerificationCodeRequest {
  email: string;
}

// 发送验证码响应
export interface SendVerificationCodeResponse {
  message: string;
}

// 注册请求
export interface RegisterWithEmailRequest {
  email: string;
  code: string;
  displayName?: string;
}

// 注册响应
export interface RegisterWithEmailResponse {
  user: {
    id: string;
    email: string;
    displayName: string | null;
    avatarUrl: string | null;
    emailVerified: boolean;
  };
  accessToken: string;
}

// 错误响应
export interface ErrorResponse {
  statusCode: number;
  message: string;
  error: string;
}
```

#### API 函数封装

```typescript
// api/auth.ts

import axios from 'axios';
import type {
  SendVerificationCodeRequest,
  SendVerificationCodeResponse,
  RegisterWithEmailRequest,
  RegisterWithEmailResponse,
} from '@/types/auth';

const API_BASE_URL = process.env.NEXT_PUBLIC_API_BASE_URL || 'http://localhost:3000';
// 注意：后端已设置全局前缀 /api，所以完整路径是 /api/auth/email/send-code

/**
 * 发送邮箱验证码
 */
export async function sendVerificationCode(
  email: string
): Promise<SendVerificationCodeResponse> {
  const response = await axios.post<SendVerificationCodeResponse>(
    `${API_BASE_URL}/api/auth/email/send-code`,
    { email },
    {
      headers: {
        'Content-Type': 'application/json',
      },
    }
  );
  return response.data;
}

/**
 * 使用邮箱和验证码注册
 */
export async function registerWithEmail(
  data: RegisterWithEmailRequest
): Promise<RegisterWithEmailResponse> {
  const response = await axios.post<RegisterWithEmailResponse>(
    `${API_BASE_URL}/api/auth/email/register`,
    data,
    {
      headers: {
        'Content-Type': 'application/json',
      },
      withCredentials: true, // 重要：包含 cookies（用于 refresh_token）
    }
  );
  return response.data;
}
```

### 3. 注册页面组件实现

#### React 示例

```typescript
// pages/register.tsx 或 components/RegisterForm.tsx

import { useState, useEffect } from 'react';
import { useRouter } from 'next/router';
import { sendVerificationCode, registerWithEmail } from '@/api/auth';
import { useAuthStore } from '@/stores/auth'; // 假设使用 Zustand

export default function RegisterPage() {
  const router = useRouter();
  const { setUser, setAccessToken } = useAuthStore();
  
  const [step, setStep] = useState<'email' | 'code' | 'success'>('email');
  const [email, setEmail] = useState('');
  const [code, setCode] = useState('');
  const [displayName, setDisplayName] = useState('');
  const [countdown, setCountdown] = useState(0);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  // 倒计时逻辑
  useEffect(() => {
    if (countdown > 0) {
      const timer = setTimeout(() => setCountdown(countdown - 1), 1000);
      return () => clearTimeout(timer);
    }
  }, [countdown]);

  // 邮箱格式验证
  const validateEmail = (email: string): boolean => {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    return emailRegex.test(email);
  };

  // 发送验证码
  const handleSendCode = async () => {
    setError('');
    
    // 前端验证
    if (!validateEmail(email)) {
      setError('请输入有效的邮箱地址');
      return;
    }

    setLoading(true);
    try {
      await sendVerificationCode(email);
      setStep('code');
      setCountdown(60); // 60秒倒计时
      // 显示成功提示
      alert('验证码已发送，请查收邮件');
    } catch (error: any) {
      if (error.response) {
        const errorMessage = error.response.data?.message || '发送验证码失败';
        setError(errorMessage);
      } else {
        setError('网络异常，请检查网络连接');
      }
    } finally {
      setLoading(false);
    }
  };

  // 注册
  const handleRegister = async (e: React.FormEvent) => {
    e.preventDefault();
    setError('');

    // 前端验证
    if (!code || code.length !== 6) {
      setError('请输入6位验证码');
      return;
    }

    if (!validateEmail(email)) {
      setError('请输入有效的邮箱地址');
      return;
    }

    setLoading(true);
    try {
      const result = await registerWithEmail({
        email,
        code,
        displayName: displayName || undefined,
      });

      // 保存用户信息和token
      setUser(result.user);
      setAccessToken(result.accessToken);
      
      // 保存token到sessionStorage（可选）
      sessionStorage.setItem('accessToken', result.accessToken);

      // 显示成功提示
      alert('注册成功！');
      
      // 跳转到首页
      router.push('/');
    } catch (error: any) {
      if (error.response) {
        const errorMessage = error.response.data?.message || '注册失败';
        setError(errorMessage);
        
        // 根据错误类型提供不同提示
        if (errorMessage.includes('验证码')) {
          alert(errorMessage + '，请重新获取验证码');
        } else if (errorMessage.includes('已被注册')) {
          alert(errorMessage + '，请直接登录');
          setTimeout(() => router.push('/login'), 2000);
        }
      } else {
        setError('网络异常，请检查网络连接');
      }
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="register-page">
      <h1>注册</h1>

      {step === 'email' && (
        <div className="email-step">
          <input
            type="email"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            placeholder="请输入邮箱地址"
            disabled={loading}
          />
          <button
            onClick={handleSendCode}
            disabled={loading || countdown > 0}
          >
            {countdown > 0 ? `重新发送(${countdown}s)` : '发送验证码'}
          </button>
          {error && <div className="error">{error}</div>}
        </div>
      )}

      {step === 'code' && (
        <form onSubmit={handleRegister} className="code-step">
          <div>
            <label>邮箱</label>
            <input type="email" value={email} disabled />
          </div>
          <div>
            <label>验证码</label>
            <input
              type="text"
              value={code}
              onChange={(e) => setCode(e.target.value.replace(/\D/g, '').slice(0, 6))}
              placeholder="请输入6位验证码"
              maxLength={6}
              disabled={loading}
            />
            <button
              type="button"
              onClick={handleSendCode}
              disabled={countdown > 0}
            >
              {countdown > 0 ? `重新发送(${countdown}s)` : '重新发送验证码'}
            </button>
          </div>
          <div>
            <label>显示名称（可选）</label>
            <input
              type="text"
              value={displayName}
              onChange={(e) => setDisplayName(e.target.value)}
              placeholder="请输入显示名称"
              maxLength={50}
              disabled={loading}
            />
          </div>
          {error && <div className="error">{error}</div>}
          <button type="submit" disabled={loading}>
            {loading ? '注册中...' : '注册'}
          </button>
        </form>
      )}

      {step === 'success' && (
        <div className="success-step">
          <p>注册成功！</p>
          <p>正在跳转...</p>
        </div>
      )}
    </div>
  );
}
```

### 4. Token 管理

#### 配置 Axios 拦截器

```typescript
// utils/axios.ts

import axios from 'axios';

const apiClient = axios.create({
  baseURL: process.env.NEXT_PUBLIC_API_BASE_URL,
  withCredentials: true, // 重要：包含 cookies
});

// 请求拦截器：自动添加 accessToken
apiClient.interceptors.request.use((config) => {
  const accessToken = sessionStorage.getItem('accessToken');
  if (accessToken) {
    config.headers.Authorization = `Bearer ${accessToken}`;
  }
  return config;
});

// 响应拦截器：处理 token 过期
apiClient.interceptors.response.use(
  (response) => response,
  async (error) => {
    if (error.response?.status === 401) {
      // Token 过期，尝试刷新
      try {
        const refreshResponse = await axios.post(
          `${process.env.NEXT_PUBLIC_API_BASE_URL || 'http://localhost:3000'}/api/auth/refresh`,
          {},
          { withCredentials: true }
        );
        const newAccessToken = refreshResponse.data.accessToken;
        sessionStorage.setItem('accessToken', newAccessToken);
        
        // 重试原请求
        error.config.headers.Authorization = `Bearer ${newAccessToken}`;
        return apiClient.request(error.config);
      } catch (refreshError) {
        // 刷新失败，跳转到登录页
        sessionStorage.removeItem('accessToken');
        window.location.href = '/login';
        return Promise.reject(refreshError);
      }
    }
    return Promise.reject(error);
  }
);

export default apiClient;
```

### 5. 状态管理（可选）

#### 使用 Zustand 示例

```typescript
// stores/auth.ts

import { create } from 'zustand';

interface User {
  id: string;
  email: string;
  displayName: string | null;
  avatarUrl: string | null;
  emailVerified: boolean;
}

interface AuthState {
  user: User | null;
  accessToken: string | null;
  setUser: (user: User) => void;
  setAccessToken: (token: string) => void;
  logout: () => void;
}

export const useAuthStore = create<AuthState>((set) => ({
  user: null,
  accessToken: null,
  setUser: (user) => set({ user }),
  setAccessToken: (token) => {
    set({ accessToken: token });
    if (token) {
      sessionStorage.setItem('accessToken', token);
    }
  },
  logout: () => {
    set({ user: null, accessToken: null });
    sessionStorage.removeItem('accessToken');
  },
}));
```

## ⚠️ 重要注意事项

### 1. Cookie 设置

- **必须设置**: `withCredentials: true` 在 axios 请求中
- **原因**: refresh_token 通过 httpOnly cookie 传递，需要包含 cookies

### 2. Token 存储

- **推荐**: 使用 `sessionStorage` 存储 accessToken
- **不推荐**: 不要使用 `localStorage`（安全性较低）
- **注意**: refresh_token 由服务器通过 cookie 管理，前端无需手动处理

### 3. 错误处理

- **邮箱格式错误**: 显示 "请输入有效的邮箱地址"
- **验证码错误**: 显示 "验证码错误，请重新输入"
- **验证码过期**: 显示 "验证码已过期，请重新获取"
- **邮箱已注册**: 显示 "该邮箱已被注册，请直接登录" 并提供登录链接
- **发送过于频繁**: 显示倒计时，禁用发送按钮

### 4. 用户体验优化

- **倒计时显示**: 发送验证码后显示 60 秒倒计时
- **自动跳转**: 注册成功后自动跳转到首页或用户中心
- **友好提示**: 提示用户检查垃圾邮件文件夹
- **加载状态**: 显示加载动画，禁用按钮防止重复提交
