## 6. API层封装

### 6.1 创建 Axios 实例

`src/api/client.ts`：

```typescript
import axios, { AxiosInstance, AxiosError, InternalAxiosRequestConfig } from 'axios';
import { API_BASE_URL } from '@/constants/config';
import { ApiResponse } from '@/types/common';

const apiClient: AxiosInstance = axios.create({
  baseURL: API_BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// 请求拦截器
apiClient.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    const token = localStorage.getItem('token');
    if (token && config.headers) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error: AxiosError) => {
    return Promise.reject(error);
  }
);

// 响应拦截器
apiClient.interceptors.response.use(
  (response) => {
    // 如果后端返回的是统一格式 { success, data }
    return response.data;
  },
  (error: AxiosError<ApiResponse>) => {
    // 处理错误
    if (error.response?.status === 401) {
      // 未授权，跳转到登录页
      localStorage.removeItem('token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

export default apiClient;
```

### 6.2 行程 API 封装示例

`src/api/trips.ts`：

```typescript
import apiClient from './client';
import type { ApiResponse } from '@/types/common';

// 行程类型定义（应该从后端类型或OpenAPI生成）
export interface Trip {
  id: string;
  destination: string;
  startDate: string;
  endDate: string;
  budgetConfig: {
    total: number;
    currency: string;
  };
  pacingConfig: {
    mobilityProfile: string;
  };
}

export interface CreateTripDto {
  destination: string;
  startDate: string;
  endDate: string;
  budgetConfig: {
    total: number;
    currency: string;
  };
  pacingConfig: {
    mobilityProfile: string;
  };
}

export const tripsApi = {
  // 获取所有行程
  getAll: (): Promise<ApiResponse<Trip[]>> => {
    return apiClient.get('/trips');
  },
  
  // 获取行程详情
  getById: (id: string): Promise<ApiResponse<Trip>> => {
    return apiClient.get(`/trips/${id}`);
  },
  
  // 创建行程
  create: (data: CreateTripDto): Promise<ApiResponse<Trip>> => {
    return apiClient.post('/trips', data);
  },
  
  // 更新行程
  update: (id: string, data: Partial<CreateTripDto>): Promise<ApiResponse<Trip>> => {
    return apiClient.put(`/trips/${id}`, data);
  },
  
  // 删除行程
  delete: (id: string): Promise<ApiResponse<void>> => {
    return apiClient.delete(`/trips/${id}`);
  },
};
```

### 7.2 自定义 Hooks（使用 React Query）

`src/hooks/useTrips.ts`：

```typescript
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { tripsApi, type Trip, type CreateTripDto } from '@/api/trips';

// 获取所有行程
export function useTrips() {
  return useQuery({
    queryKey: ['trips'],
    queryFn: async () => {
      const response = await tripsApi.getAll();
      return response.data || [];
    },
  });
}

// 获取单个行程
export function useTrip(id: string) {
  return useQuery({
    queryKey: ['trips', id],
    queryFn: async () => {
      const response = await tripsApi.getById(id);
      return response.data;
    },
    enabled: !!id,
  });
}

// 创建行程
export function useCreateTrip() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: (data: CreateTripDto) => tripsApi.create(data),
    onSuccess: () => {
      // 刷新行程列表
      queryClient.invalidateQueries({ queryKey: ['trips'] });
    },
  });
}

// 更新行程
export function useUpdateTrip() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: ({ id, data }: { id: string; data: Partial<CreateTripDto> }) =>
      tripsApi.update(id, data),
    onSuccess: (_, variables) => {
      queryClient.invalidateQueries({ queryKey: ['trips'] });
      queryClient.invalidateQueries({ queryKey: ['trips', variables.id] });
    },
  });
}

// 删除行程
export function useDeleteTrip() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: (id: string) => tripsApi.delete(id),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['trips'] });
    },
  });
}
```

```typescript
interface ProtectedRouteProps {
  children: React.ReactNode;
}

export function ProtectedRoute({ children }: ProtectedRouteProps) {
  const location = useLocation();
  const token = localStorage.getItem('token');
  
  if (!token) {
    return <Navigate to="/login" state={{ from: location }} replace />;
  }
  
  return <>{children}</>;
}
```

### 10.2 创建行程页面

`src/pages/trips/TripNewPage.tsx`：

```typescript
import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { useCreateTrip } from '@/hooks/useTrips';
import type { CreateTripDto } from '@/api/trips';

export default function TripNewPage() {
  const navigate = useNavigate();
  const createTrip = useCreateTrip();
  const [formData, setFormData] = useState<CreateTripDto>({
    destination: '',
    startDate: '',
    endDate: '',
    budgetConfig: {
      total: 0,
      currency: 'CNY',
    },
    pacingConfig: {
      mobilityProfile: 'STAMINA_60_TERRAIN_NO_STAIRS',
    },
  });

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    try {
      const response = await createTrip.mutateAsync(formData);
      if (response.data) {
        navigate(`/trips/${response.data.id}`);
      }
    } catch (error) {
      console.error('创建行程失败:', error);
    }
  };
```

### 13.2 错误边界

`src/components/common/ErrorBoundary.tsx`：

```typescript
import React, { Component, ErrorInfo, ReactNode } from 'react';

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
  error?: Error;
}

class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('Error caught by boundary:', error, errorInfo);
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="error-boundary">
          <h2>出现错误</h2>
          <p>{this.state.error?.message}</p>
          <button onClick={() => this.setState({ hasError: false })}>
            重试
          </button>
        </div>
      );
    }

    return this.props.children;
  }
}

export default ErrorBoundary;
```

## 18. 故障排查

### 18.1 常见错误及解决方法

**错误1: Cannot find module '@/*'**

**原因**: TypeScript 路径别名未配置

**解决**: 检查 `tsconfig.json` 和 `vite.config.ts` 中的路径配置

**错误2: React Hook useEffect has a missing dependency**

**原因**: useEffect 依赖项不完整

**解决**: 
```typescript
// ❌ 错误
useEffect(() => {
  fetchData(id);
}, []); // 缺少 id 依赖

// ✅ 正确
useEffect(() => {
  fetchData(id);
}, [id]);
```

**错误3: Maximum update depth exceeded**

**原因**: 在渲染过程中触发了状态更新

**解决**: 确保状态更新不在 render 函数中直接调用

**错误4: CORS 错误**

**原因**: 后端未配置 CORS 或代理配置错误

**解决**: 
- 检查后端 CORS 配置
- 检查 Vite proxy 配置
