```typescript
// 完整流程测试
describe('助手对话界面 - 完整流程', () => {
  it('应该能够填充空闲时间并一键添加', async () => {
    // 1. 发送填充空闲时间请求
    const chatResponse = await chat({
      tripId: 'xxx',
      message: '帮我填充空闲时间',
      sessionId: 'xxx',
    });
    
    // 2. 验证返回推荐列表
    expect(chatResponse.richContent?.type).toBe('poi_list');
    expect(chatResponse.richContent?.data.recommendations.length).toBeGreaterThan(0);
    
    // 3. 选择第一个推荐，一键添加
    const recommendation = chatResponse.richContent.data.recommendations[0];
    const suggestion = recommendation.suggestions[0];
    
    const addResponse = await applySuggestion({
      tripId: 'xxx',
      sessionId: 'xxx',
      suggestionId: suggestion.id,
      targetDay: recommendation.day,
      timeSlot: recommendation.timeSlot,
      suggestionType: suggestion.type === 'RESTAURANT' ? 'add_meal' : 'add_place',
      place: {
        name: suggestion.name,
        category: suggestion.type,
      },
    });
    
    // 4. 验证添加成功
    expect(addResponse.success).toBe(true);
    expect(addResponse.item).toBeDefined();
  });
  
  it('应该能够接受Abu建议并更新时间', async () => {
    // 1. 获取Abu建议
    const chatResponse = await chat({
      tripId: 'xxx',
      message: '查看行程',
      sessionId: 'xxx',
    });
    
    // 2. 找到Abu的时间警告
    const abuWarning = chatResponse.personaInsights?.find(
      insight => insight.persona === 'Abu' && insight.message.includes('太早')
    );
    
    expect(abuWarning).toBeDefined();
    
    // 3. 接受建议
    const applyResponse = await applySuggestion({
      tripId: 'xxx',
      sessionId: 'xxx',
      suggestionId: abuWarning.id,
      targetDay: 1,
      suggestionType: 'modify_time',
    });
    
    // 4. 验证时间已调整
    expect(applyResponse.success).toBe(true);
    expect(applyResponse.suggestionStatus).toBe('RESOLVED');
    expect(applyResponse.item.startTime).not.toContain('01:00');
    
    // 5. 重新获取行程，验证警告已消失
    const updatedResponse = await chat({
      tripId: 'xxx',
      message: '查看行程',
      sessionId: 'xxx',
    });
    
    const updatedWarning = updatedResponse.personaInsights?.find(
      insight => insight.persona === 'Abu' && insight.message.includes('太早')
    );
    
    expect(updatedWarning).toBeUndefined();
  });
});
```
